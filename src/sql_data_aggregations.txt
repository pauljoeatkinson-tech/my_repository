/*-- create employee table from input csv file --*/
CREATE EXTERNAL TABLE employee (
  Age int,
  Attrition string,
  BusinessTravel string,
  DailyRate int,
  Department string,
  DistanceFromHome int,
  Education int,
  EducationField string,
  EmployeeCount int,
  EmployeeNumber int,
  EnvironmentSatisfaction int,
  Gender string,
  HourlyRate int,
  JobInvolvement int,
  JobLevel int,
  JobRole string,
  JobSatisfaction int,
  MaritalStatus string,
  MonthlyIncome int,
  MonthlyRate int,
  NumCompaniesWorked int,
  Ovr18 string,
  OvrTime string,
  PercentSalaryHike int,
  PerformanceRating int,
  RelationshipSatisfaction int,
  StandardHours int,
  StockOptionLevel int,
  TotalWorkingYears int,
  TrainingTimesLastYear int,
  WorkLifeBalance int,
  YearsAtCompany int,
  YearsInCurrentRole int,
  YearsSinceLastPromotion int,
  YearsWithCurrManager int
)
ROW FORMAT DELIMITED 
FIELDS TERMINATED BY ',' 
LINES TERMINATED BY '\n' 
STORED AS TEXTFILE 
LOCATION 'wasbs://employee@employeehdistorage.blob.core.windows.net/employee' 
TBLPROPERTIES("skip.header.line.count"="1");


/*-- count of sales people with attrition --*/
CREATE TABLE Attrition AS
SELECT COUNT(*) AS Attrition
FROM employee
WHERE Attrition = 'Yes' AND Department = 'Sales';


/*-- count of sales people without attrition --*/
CREATE TABLE Non_Attrition AS
SELECT COUNT(*) AS Non_Attrition
FROM employee
WHERE Attrition = 'No' AND Department = 'Sales';


/*-- average monthly income for all sales people --*/
CREATE TABLE Table_Average AS 
SELECT AVG(MonthlyIncome) AS average_monthly_income
FROM employee
WHERE Department = 'Sales';


/* -- minimum monthly income for all sales people --*/
CREATE TABLE Table_Min AS 
SELECT MIN(MonthlyIncome) AS min_monthly_income
FROM employee
WHERE Department = 'Sales';


/* -- maximum monthly income for all sales people --*/
CREATE TABLE Table_Max AS 
SELECT MAX(MonthlyIncome) AS max_monthly_income
FROM employee
WHERE Department = 'Sales';


/*-- distinct monthly incomes of sales people with attrition, sorted by count --*/
CREATE TABLE Sales_Attrition AS 
SELECT 
    MonthlyIncome AS Monthly_Income, 
    COUNT(*) AS count
FROM 
    employee 
WHERE 
    Attrition = 'Yes' 
    AND Department = 'Sales'
GROUP BY 
    MonthlyIncome
ORDER BY 
count DESC;


/*-- distinct monthly incomes of sales people without attrition, sorted by count --*/
CREATE TABLE Sales_Non_Attrition AS 
SELECT 
    MonthlyIncome AS Monthly_Income, 
    COUNT(*) AS count 
FROM 
    employee 
WHERE 
    Attrition = 'No' 
    AND Department = 'Sales'
GROUP BY 
    MonthlyIncome
ORDER BY 
count DESC;
